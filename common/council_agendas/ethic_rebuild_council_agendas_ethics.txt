@base_ethic_agenda_cost = 7000

agenda_environmental_stewards = {
	agenda_cost = @base_ethic_agenda_cost
	potential = {
		is_gestalt = no
		is_nomadic = no
	}
	allow = {
		custom_tooltip = {
			fail_text = "requires_green"
			is_green = yes
		}
	}
	modifier = {
		deposit_blockers_cost_mult = -0.33
		pop_citizen_happiness = 0.05
		pop_ethic_green_attraction_mult = 0.1
	}
	finish_modifier = agenda_environmental_stewards_finish
	effect = {
		# Add one random natural blocker to, or removes an industrial wasteland from, 3 random colonies we own. 
		# If this cannot be done, gives the planet the Natural Beauty modifier. 
		# If that's already been done, boost society research from biologists for 720 days.
		# Yay, go green! -- Sage
		custom_tooltip = council_agenda_agenda_environmental_stewards_effect
	
		add_environmental_steward_agenda = yes
		if = {
				limit = {
					any_situation = {
						is_situation_type = paragon_origin_promise
					}
					has_country_flag = origin_upgrade_agendas_situation
					has_overlord_dlc = yes
				}
				country_event = {
					id = paragon.5305
				}
				hidden_effect = {
					country_event = {
						id = paragon.5014
						days = 180
						random = 360
					}
				}
		}
		else_if = {
				limit = {
					any_situation = {
						is_situation_type = paragon_origin_promise
					}
					has_country_flag = origin_upgrade_agendas_situation
					has_overlord_dlc = no
				}
				country_event = {
					id = paragon.5320
				}
				hidden_effect = {
					country_event = {
						id = paragon.5014
						days = 180
						random = 360
					}
				}
		}
	}
}
