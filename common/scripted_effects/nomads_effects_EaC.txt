# Performs a population exchange between event_target:migrating_arkship and event_target:migrating_planet,
# applies the per-host-planet 5-year cooldown, and notifies the host empire.
# Callers must save migrating_planet and migrating_arkship before invoking.
arkship_migration_effect = {
	event_target:migrating_planet = {
		set_timed_carrier_flag = {
			flag = migration_happened_planet
			years = 5
		}
		owner = {
			create_message = {
				type = MESSAGE_NOMAD_MIGRATION_HOST
				localization = MESSAGE_NOMAD_MIGRATION_HOST_DESC
				days = 30
				target = root
				variable = {
					type = name
					localization = COUNTRY
					scope = root
				}
				variable = {
					type = name
					localization = PLANET
					scope = event_target:migrating_planet
				}
				variable = {
					type = name
					localization = ARKSHIP
					scope = event_target:migrating_arkship
				}
			}
		}
		random_owned_pop_group = {
			resettle_pop_group = {
				POP_GROUP = this
				PLANET = event_target:migrating_arkship
				AMOUNT = value:compare_resettle_attraction|ORIGIN|event_target:migrating_planet|DEST|event_target:migrating_arkship|
			}
		}
	}
	event_target:migrating_arkship = {
		random_owned_pop_group = {
			resettle_pop_group = {
				POP_GROUP = this
				PLANET = event_target:migrating_planet
				AMOUNT = value:compare_resettle_attraction|ORIGIN|event_target:migrating_arkship|DEST|event_target:migrating_planet|
			}
		}
	}
	# Forever Cruise: set new migrant species to passenger citizenship
	if = {
		limit = {
			event_target:migrating_arkship.owner = {
				has_origin = origin_forever_cruise
				NOT = { has_country_flag = forever_cruise_over }
				is_hospitality_servitor_empire = no
				is_hospitality_machine_cult_machine_empire = no
			}
		}
		event_target:migrating_arkship.owner = {
			every_owned_species = {
				limit = {
					NOR = {
						is_same_species = prev
						has_citizenship_type = { type = citizenship_cruise_crew country = prev }
						has_citizenship_type = { type = citizenship_cruise_passenger country = prev }
					}
				}
				set_citizenship_type = {
					type = citizenship_cruise_passenger
					country = prev
				}
				set_living_standard = {
					type = living_standard_cruise_passenger
					country = prev
				}
			}
		}
	}
	# Hospitality Servitor: set new migrant species to organic trophy
	else_if = {
		limit = {
			event_target:migrating_arkship.owner = {
				OR = {
					is_hospitality_servitor_empire = yes
					is_hospitality_machine_cult_machine_empire = yes
				}
			}
		}
		event_target:migrating_arkship.owner = {
			every_owned_species = {
				limit = {
					NOR = {
						is_same_species = prev
						has_citizenship_type = { type = citizenship_cruise_crew country = prev }
						has_citizenship_type = { type = citizenship_organic_trophy country = prev }
					}
				}
				if = {
					limit = {
						NOR = {
							has_trait = trait_mechanical
							has_trait = trait_machine_unit
						}
					}
					set_citizenship_type = {
						type = citizenship_organic_trophy
						country = prev
					}
					set_living_standard = {
						type = living_standard_organic_trophy
						country = prev
					}
				}
				else = {
					set_citizenship_type = {
						type = citizenship_cruise_crew
						country = prev
					}
				}
			}
		}
	}
}

planetary_pilgrimage_scaling_reward_effect = {
	if = {
		limit = {
			is_variable_set = planetary_pilgrimage_completed
		}
		root = {
			if = {
				limit = { has_country_flag = contract_downtime_taken }
				add_resource = {
					unity = $UNITY$
					mult = value:contract_bonus_value_penalized
				}
				hidden_effect = { country_event = { id = nomads.6998 days = 1 } }
				hidden_effect = { country_event = { id = EaC_civic_mechanic_events.5006 days = 1 } }
			}
			else = {
				add_resource = {
					unity = $UNITY$
					mult = value:contract_bonus_value
				}
				add_resource = {
					$RESOURCE$ = $RESOURCE_AMOUNT$
					mult = value:contract_bonus_value
				}
			}
		}
	}
}
