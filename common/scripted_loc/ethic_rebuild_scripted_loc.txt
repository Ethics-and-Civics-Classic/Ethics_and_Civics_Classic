defined_text = {
	name = GetSlotCity02Prereq
	text = {
		trigger = {
			exists = owner
			owner = {
				NOT = { has_technology = tech_primitive_unified_planet }
			}
		}
		localization_key = zone_unavailable_tech_primitive_unified_planet
	}
	text = {
		trigger = {
			exists = owner
			owner = {
				NOT = { has_technology = tech_colonial_centralization }
				has_technology = tech_primitive_unified_planet
			}
		}
		localization_key = zone_unavailable_tech_colonial_centralization
	}
	text = {
		trigger = {
			exists = owner
			owner = {
				has_technology = tech_colonial_centralization
				has_technology = tech_primitive_unified_planet
			}
			has_upgraded_capital = no
		}
		localization_key = zone_unavailable_capital
	}
}

		trigger = {

}
