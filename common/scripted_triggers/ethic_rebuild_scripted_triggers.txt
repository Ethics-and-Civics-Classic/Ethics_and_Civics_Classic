	OR = {
		any_system_within_border = {
			has_star_flag = here_be_dragons_system
			any_fleet_in_system = {
				is_ship_size = space_dragon_origin
				owner = {
					has_country_flag = here_be_the_dragon@prevprevprev
					NOT = { is_hostile = prevprevprev }
				}
			}
		}

		capital_scope.solar_system = {
			has_star_flag = here_be_dragons_system
			any_fleet_in_system = {
				is_ship_size = space_dragon_origin
				if = {
					limit = { root = { is_scope_type = country } }
					owner = {
						has_country_flag = here_be_the_dragon@root
						NOT = { is_hostile = root }
					}
				}
				else = {
					owner = {
						has_country_flag = here_be_the_dragon@root.owner
						NOT = { is_hostile = root.owner }
					}
				}
			}
		}
	}
}
