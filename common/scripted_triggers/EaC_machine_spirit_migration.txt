eac_machine_spirit_category_candidate = {
	pop_group_size > 0
	exists = owner
	is_robotic_species = no
	is_being_purged = no
	is_being_assimilated = no
	colony? = { NOT = { is_planet_class = pc_cosmogenesis_world } }
}

eac_machine_spirit_legacy_job_group = {
	eac_machine_spirit_category_candidate = yes
	owner = { has_valid_civic = civic_machine_cult_machine }
	NOR = {
		is_pop_category = bio_trophy
		is_pop_category = bio_trophy_unemployment
	}
	# Civilian rights are safe only for the amount demonstrably assigned to the
	# old Specialist Techno Priest. The original processing bridge stays active.
	OR = {
		has_citizenship_type = {
			type = citizenship_organic_trophy
			country = owner
		}
		has_citizenship_type = {
			type = citizenship_organic_civilian
			country = owner
		}
	}
}

eac_machine_spirit_rights_away_group = {

			country = owner
		}

		}
	}
