entertainer_is_storm_interpreter_trigger = {
	exists = owner
	OR = {
		owner = {
			is_storm_callers_empire = yes
		}
		AND = {
			owner = {
				OR = {
					is_green = yes
					is_industrial = yes
				}
			}
			exists = planet
			planet = {
				solar_system? = {
					OR = {
						has_environmental_effects = yes
						has_star_flag = storm_system
					}
				}
			}
		}
	}
}

entertainer_job_check_trigger = {
	custom_tooltip = SPECIALIST_JOB_TRIGGER
	hidden_trigger = {
		exists = owner
		OR = {
			is_enslaved = no
			has_slavery_type = { type = slavery_domestic }
		}
		NOR = { is_being_purged = yes is_being_assimilated = yes has_trait = trait_syncretic_proles }
		NOT = { has_trait = trait_slave_gene }
		can_think = yes
		# Rule out Disconnected Drones
		has_disconnected_drone_citizenship_type = no
		OR = {
			NOT = { has_trait = trait_mechanical }
			owner = { has_technology = tech_droid_workers }
		}
		NOT = {
			has_citizenship_type = {
				type = citizenship_organic_trophy
				country = owner
			}
		}
		if = {
			limit = {
				divinity_right_to_work_job_check_trigger_exempt = no
			}
			divinity_right_to_work_job_check_trigger_specialist = yes
		}
	}
}
