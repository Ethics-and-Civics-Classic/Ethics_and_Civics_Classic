### Regular Empires
# Planets
triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
			is_fallen_empire_spiritualist = no
			is_nomadic = no
			has_technology = tech_industrial_farming_2
		}
		planet_is_habitat_equivalent = no
		planet_is_ecu_equivalent = no
		planet_is_ring_world_equivalent = no
	}
	job_farmer_add = $AMOUNT$
}
#Primitive

	potential = {

			is_regular_empire = yes
			is_fallen_empire_spiritualist = no

			NOT = { has_technology = tech_industrial_farming_2 }
		}
		planet_is_habitat_equivalent = no
		planet_is_ecu_equivalent = no
		planet_is_ring_world_equivalent = no
	}
	modifier = {
		job_primitive_farmer_add = $AMOUNT$
	}
}
