

triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
			is_fallen_empire_spiritualist = no
			OR = {
				has_technology = "tech_fusion_power"
				NOT = { any_situation = { is_situation_type = PrimPlayer_situation } }
			}
		}
		planet_is_habitat_equivalent = no
		planet_is_ecu_equivalent = no
		planet_is_ring_world_equivalent = no
	}
	job_technician_add = $AMOUNT$
}
