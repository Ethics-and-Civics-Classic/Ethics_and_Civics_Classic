# Regular Empires
triggered_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
			is_fallen_empire_spiritualist = no
			is_gestalt = no
			is_nomadic = no
			has_technology = tech_industrial_farming_2
		}
	}
	modifier = {
		job_farmer_add = $AMOUNT$
	}
}
