# Synthetics Prefabricated Components
# +100 Roboticist
triggered_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_regular_empire = yes
			has_active_tradition = tr_synthetics_prefabricated_components
		}
	}
	job_roboticist_add = 100
}

# Individualist Machines
# +200 Roboticist

	potential = {
		exists = owner
		owner = {
			is_individual_machine = yes
		}
	}
	job_roboticist_add = 200
}

# Pharma State
# +100 Medical Worker
triggered_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			OR = {
				has_valid_civic = civic_private_healthcare_corporate
				has_valid_civic = civic_welfare_state
			}
		}
	}
	job_healthcare_add = 100
}

# Healthcare Jobs (Enlightment)
triggered_planet_modifier = {
	potential = {
		capital_tier >= 2
		exists = owner
		owner = {
			has_valid_civic = civic_enlightment
			NOT = {
				prev = { capital_tier >= 3 }
			}
		}
	}
	job_healthcare_add = 100
}
