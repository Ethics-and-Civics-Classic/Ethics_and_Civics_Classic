
modifier = {
	factor = @cruise_contract_mult
}

modifier = {
	factor = 0
	has_country_flag = contract_planetary_pilgrimage_$VARIANT$_issued
}

modifier = {
	factor = 0
	has_country_flag = planetary_pilgrimage_failed
}

random_owned_fleet = {
	limit = {
		is_arkship_fleet = yes
	}
	solar_system = {
		save_event_target_as = arkship_system
	}
}

random_system = {
	limit = {
		distance = {
			source = event_target:arkship_system
			min_jumps = 1
			max_jumps = 20
		}
		root = {
			intel_level = {
				level > none
				system = prev
			}
		}
		OR = {
			is_pilgrimage_site_space_fauna = yes
			any_system_planet = {
				OR = {
					is_pilgrimage_site_default = yes
					is_pilgrimage_site_wildlife = yes
					is_pilgrimage_site_holy_world = yes
					is_pilgrimage_site_presapient = yes
				}
			}
		}
		NOT = {
			has_star_flag = planetary_pilgrimage_contract_issued@root
		}
	}
	save_event_target_as = contract_system
	event_target:nomad_country = {
		issue_contract = {
			contract = contract_planetary_pilgrimage_$VARIANT$
			location = event_target:contract_system
			target = root
		}
		set_country_flag = planetary_pilgrimage_issued
		set_timed_country_flag = {
			flag = contract_planetary_pilgrimage_$VARIANT$_issued
			days = 720
		}
	}
}

if = {
	limit = {
		NOT = { has_country_flag = planetary_pilgrimage_issued }
	}
	set_country_flag = planetary_pilgrimage_failed
}
