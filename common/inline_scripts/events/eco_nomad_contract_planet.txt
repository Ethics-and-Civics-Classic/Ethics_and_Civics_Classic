
modifier = {
	factor = 0
	has_country_flag = contract_$CONTRACT_NAME$_$VARIANT$_issued
}

modifier = {
	factor = 0
	has_country_flag = $CONTRACT_NAME$_failed
}

random_owned_fleet = {
	limit = {
		is_arkship_fleet = yes
	}
	solar_system = {
		save_event_target_as = arkship_system
	}
}

random_system = {
	limit = {
		distance = {
			source = event_target:arkship_system
			min_jumps = 1
			max_jumps = 10
		}
		root = {
			intel_level = {
				level > none
				system = prev
			}
		}
		any_system_planet = {
			is_$SITE$_$TYPE$ = yes
		}
		NOT = {
			has_star_flag = $CONTRACT_NAME$_contract_issued@root
		}
	}
	save_event_target_as = contract_system
	event_target:nomad_country = {
		issue_contract = {
			contract = contract_$CONTRACT_NAME$_$VARIANT$
			location = event_target:contract_system
			target = root
		}
		set_country_flag = $CONTRACT_NAME$_issued
		set_timed_country_flag = {
			flag = contract_$CONTRACT_NAME$_$VARIANT$_issued
			days = 720
		}
	}
}

if = {
	limit = {
		NOT = { has_country_flag = $CONTRACT_NAME$_issued }
	}
	set_country_flag = $CONTRACT_NAME$_failed
}
