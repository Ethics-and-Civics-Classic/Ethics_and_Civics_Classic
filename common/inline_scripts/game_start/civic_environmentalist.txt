if = {
	limit = {
		owner? = {
			has_valid_civic = civic_environmentalist
		}
		OR = {
			AND = {
				is_regular_empire = yes
				is_nomadic = no
				NOR = {
					is_planet_class = pc_city
					is_planet_class = pc_hive
					is_planet_class = pc_machine
					is_planet_class = pc_relic
					is_artificial = yes
				}
			}
			AND = {
				is_regular_empire = yes
				is_nomadic = yes
			}
		}
	}
	if = {
		limit = {
			owner? = {
				has_origin = origin_cybernetic_creed
			}
		}
		add_building = building_ranger_lodge
		add_blocker = { type = d_big_nature_preserve_blocker }
		every_deposit = {
			limit = {
				is_deposit_type = d_failing_infrastructure
			}
			remove_deposit = yes
			prev = { add_random_natural_blocker = yes }
		}
	}
	else_if = {
		limit = {
			owner? = {
				is_nomadic = yes
				has_building = building_nomad_conservatory
			}
		}
		every_deposit = {
			limit = {
				is_deposit_type = d_decrepit_dwellings
			}
			clear_blocker = yes
			prev = { add_blocker = { type = d_ark_nature_preserve_blocker } }
		}
		add_research_option = tech_alien_life_studies
	}
	else = {
		add_building = {
			zone = zone_research_unity
			building = building_ranger_lodge
		}
		add_blocker = { type = d_big_nature_preserve_blocker }
		every_deposit = {
			limit = {
				is_deposit_type = d_failing_infrastructure
			}
			remove_deposit = yes
			prev = { add_random_natural_blocker = yes }
		}
	}
}
