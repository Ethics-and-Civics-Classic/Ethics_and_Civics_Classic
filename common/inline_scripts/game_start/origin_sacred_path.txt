# The Sacred Path Initial Setup
# Zones:
#	* Replaces district_ark_city slot 1 with zone_ark_sacred_path_pilgrims (non-gestalt only)
#   * Adds the Conservatory and Ark Gardens if Ecocentrist.

if = {
	limit = {
		owner? = {
			has_origin = origin_sacred_path
		}
	}

	if = {
		limit = {
			owner = {
				is_gestalt = no
			}
		}
		add_zone = {
			district = district_ark_city
			zone = zone_ark_sacred_path_pilgrims
			zone_slot = 1
			replace = yes
		}
	}

	if = {
		limit = { 
			owner = {
				is_green = yes
			} 
		}
		add_building = {
			zone = zone_ark_sacred_path_pilgrims
			building = building_nomad_conservatory
		}
		add_blocker = { type = d_ark_nature_preserve_blocker }
	}
}
