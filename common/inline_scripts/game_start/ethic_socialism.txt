# Socialist Ethics Initial Setup
# Buildings Added:
#  * Medical Center if Biological
#  * Alloy Foundry if Machine
#  * Additional Factory if country uses Consumer Goods

if = {
	limit = {
		owner? = {
			is_socialism = yes
			is_individual_machine = no
			NOT = { has_valid_civic = civic_private_healthcare_corporate }
		}
	}
	add_building_if_not_present = {
		BUILDING = building_medical_1
	}
}

if = {
	limit = {
		owner? = {
			is_socialism = yes
			is_individual_machine = yes
		}
	}
	add_building_if_not_present = {
		BUILDING = building_foundry_1
	}
}

if = {
	limit = {
		owner? = {
			is_socialism = yes
			country_uses_consumer_goods_game_start = yes
		}
		add_building = {
			BUILDING = building_factory_1
		}
	}
}

if = {
	limit = {
		owner? = {
			is_socialism = yes
			OR = {
				has_valid_civic = civic_shared_burden
				has_valid_civic = civic_anarchy_utopia
				has_valid_civic = civic_anarchy_ideals
			}
		}
	}
	add_building_if_not_present = {
		BUILDING = building_communal_housing_large
	}
}
