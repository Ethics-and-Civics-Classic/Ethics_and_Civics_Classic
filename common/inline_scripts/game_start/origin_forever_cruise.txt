# Forever Cruise Initial Setup
# Urban Districts:
#	* 2× Forever Cruise Districts
# Generator Districts:
#	* -1 Generator Districts
# Zones:
#	* Replaces district_ark_city slot 2 with zone_ark_forever_cruise_city_mixed
# Buildings Added:
#	* Passenger Dorms

if = {
	limit = {
		owner? = {
			has_origin = origin_forever_cruise
		}
	}

	remove_district = district_ark_city
	add_district = district_ark_forever_cruise
	add_district = district_ark_forever_cruise

	add_zone = {
		district = district_ark_city
		zone = zone_ark_forever_cruise_city_mixed
		zone_slot = 2
		replace = yes
	}
	change_variable = {
		which = energy_districts
		value = -1
	}
	if = {
		limit = {
			owner = { 
				OR = {
					is_hospitality_servitor_empire = yes 
					is_hospitality_machine_cult_machine_empire = yes
				}
			}
		}
		add_building = {
			building = building_passenger_dorms
			zone = zone_ark_forever_cruise_city_mixed
		}
	}
	add_building = {
		building = building_passenger_dorms
		zone = zone_ark_forever_cruise_city_mixed
	}
	if = {
		limit = {
			owner = {
				is_hive_empire_with_machines = yes
			}
		}
		add_building = {
			building = building_machine_assembly_plant
			zone = zone_ark_forever_cruise_city_mixed
		}
	}
	if = {
		limit = {
			owner = {
				is_regular_empire = yes
				is_individual_machine = no
				has_machine_subspecies = yes
			}
		}
		add_building = {
			building = building_robot_assembly_plant
			zone = zone_ark_forever_cruise_city_mixed
		}
	}
}
