# Conservation Sites for Environmentalists
	category = $CATEGORY$
	desc = "contract_conservation_study_$VARIANT$_desc"
	lore_issued = "contract_conservation_study_$VARIANT$_lore_issued"
	lore_completed = "contract_conservation_study_$VARIANT$_lore_completed"
	lore_failed = "contract_conservation_study_$VARIANT$_lore_failed"
	event_chain = contract_conservation_study_$VARIANT$_chain
	icon = gfx/interface/icons/situation_log/situation_log_quest.dds
	picture = $PICTURE$
	location = yes
	time_to_accept = @contract_time_base

	possible_operator = {
		custom_tooltip = {
			fail_text = already_doing_same_contract
			NOT = {
				has_event_chain = contract_conservation_study_$VARIANT$_chain
			}
		}
		contract_target_system_reachable = {
			SYSTEM = from
		}
	}

	potential_issuer = {
		always = no
	}

	abort_trigger = {
		NAND = {
			has_event_chain = contract_conservation_study_$VARIANT$_chain
			contract_target_system_reachable = {
				SYSTEM = fromfrom
			}
		}
		# In the event that the planet in question is no longer a valid conservation site,
		# this trigger will abort the contract. Say, for example, the pre-sapients have been
		# uplifted, or the Tiyanki have been driven to extinction.
		fromfrom = {
            any_system_planet = {
                is_conservation_site_$STUDY$ = no
                is_point_of_interest = {
                    id = conservation_poi@this
                    event_chain = contract_conservation_study_$VARIANT$_chain
                    owner = root
                }
            }
        }
	}

	counter = {
		contract_progress_counter = {
			max = 1
		}
	}

	on_issue = {
		fromfrom = {
			set_star_flag = conservation_study_contract_issued@root
		}
	}

	on_accept = {
		# Start the chain.
		begin_event_chain = {
			event_chain = contract_conservation_study_$VARIANT$_chain
			target = ROOT
		}
		fromfrom = {
			# Find a conservation site.
			random_system_planet = {
				limit = {
					# Is a valid target?
					is_conservation_site_$STUDY$ = yes
				}
				# Mark it as a PoI.
				root = {
					create_point_of_interest = {
						id = conservation_poi@prev
						name = "conservation_poi_name"
						desc = "conservation_poi_desc"
						event_chain = contract_conservation_study_$VARIANT$_chain
						location = prev
					}
				}
				# Set the project.
				enable_special_project = {
					name = "CONTRACT_CONSERVATION_$VARIANT$_PROJECT"
					location = prev
					owner = root
				}
			}
		}
	}

	on_stop = {
		if = {
            limit = {
                exists = root
            }
			# Stop the chain.
			end_event_chain = contract_conservation_study_$VARIANT$_chain
			fromfrom = {
				# Remove the star flag.
				remove_star_flag = conservation_study_contract_issued@root
				# And the PoI marker.
				every_system_planet = {
					root = {
						remove_point_of_interest = conservation_poi@prev
					}
				}
			}
		}
	}
