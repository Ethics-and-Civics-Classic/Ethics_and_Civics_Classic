# Pilgrimage Sites for Keepers of Harmony
	category = $CATEGORY$
	desc = "contract_planetary_pilgrimage_$VARIANT$_desc"
	lore_issued = "contract_planetary_pilgrimage_$VARIANT$_lore_issued"
	lore_completed = "contract_planetary_pilgrimage_$VARIANT$_lore_completed"
	lore_failed = "contract_planetary_pilgrimage_$VARIANT$_lore_failed"
	event_chain = contract_planetary_pilgrimage_$VARIANT$_chain
	icon = gfx/interface/icons/situation_log/situation_log_quest.dds
	picture = $PICTURE$
	location = yes
	time_to_accept = @contract_time_base

	possible_operator = {
		custom_tooltip = {
			fail_text = already_doing_same_contract
			NOT = {
				has_event_chain = contract_planetary_pilgrimage_$VARIANT$_chain
			}
		}
		contract_target_system_reachable = {
			SYSTEM = from
		}
	}

	potential_issuer = {
		always = no
	}

	abort_trigger = {
		NAND = {
			has_event_chain = contract_planetary_pilgrimage_$VARIANT$_chain
			contract_target_system_reachable = {
				SYSTEM = fromfrom
			}
		}
	}

	counter = {
		contract_progress_counter = {
			max = 1
		}
	}

	on_issue = {
		fromfrom = {
			set_star_flag = planetary_pilgrimage_contract_issued@root
		}
	}

	on_accept = {
		begin_event_chain = {
			event_chain = contract_planetary_pilgrimage_$VARIANT$_chain
			target = ROOT
		}
		fromfrom = {
			random_system_planet = {
				limit = {
					is_pilgrimage_site_$WORSHIP$ = yes
				}
				save_event_target_as = contract_planet
			}
		}
		enable_special_project = {
			name = "CONTRACT_PILGRIMAGE_$VARIANT$_PROJECT"
			location = fromfrom
			owner = root
		}
	}

	on_stop = {
		end_event_chain = contract_planetary_pilgrimage_$VARIANT$_chain
		fromfrom = {
			remove_star_flag = planetary_pilgrimage_contract_issued@root
		}
	}
