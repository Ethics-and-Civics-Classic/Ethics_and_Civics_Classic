# Conservation Study
	key = "CONTRACT_CONSERVATION_$VARIANT$_PROJECT"
	event_chain = contract_conservation_study_$VARIANT$_chain
	days_to_research = 180
	project_type = project_type_contract
	picture = $PICTURE$
	icon = "gfx/interface/icons/situation_log/situation_log_quest.dds"
	location = yes
	remove_when_completed = yes

	event_scope = ship_event

	requirements = {
		carries_colony = 1
	}

	on_success = {
		owner = {
			add_mission_counter = {
				mission = contract_conservation_study_$VARIANT$
				counter = contract_progress_counter
				amount = 1
			}
		}
	}

	on_fail = {
		owner = { end_event_chain = contract_conservation_study_$VARIANT$_chain }
	}

	abort_trigger = {
		NOT = {
			owner = { has_event_chain = contract_conservation_study_$VARIANT$_chain }
		}
	}

	on_cancel = {
		# Do nothing
	}
