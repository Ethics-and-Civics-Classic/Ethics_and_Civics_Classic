## Default Pilgrimage
contract_planetary_pilgrimage_1 = {
	inline_script = {
		script = contracts/PlanetaryPilgrimage
		CATEGORY = planetary_pilgrimage_special
		VARIANT = 1
		WORSHIP = default
		PICTURE = GFX_evt_astral_rift_psionic_stranger
	}

	on_success = {
		if = {
			limit = {
				exists = root
			}
			planetary_pilgrimage_scaling_reward_effect = {
				UNITY = 1000
				RESOURCE = rare_crystals
				RESOURCE_AMOUNT = 100
			}
			hidden_effect = {
				country_event = { id = EaC_civic_mechanic_events.5006 days = 1 }
				country_event = { id = nomads.6998 days = 1 }
			}
		}
	}

	on_cancel = {
		if = {
			limit = { has_event_chain = contract_planetary_pilgrimage_1_chain }
			end_event_chain = contract_planetary_pilgrimage_1_chain
		}
		if = {
			limit = {
				has_special_project = CONTRACT_PILGRIMAGE_1_PROJECT
			}
			abort_special_project = {
				type = CONTRACT_PILGRIMAGE_1_PROJECT
			}
		}
	}
}

# Space Fauna Home System Pilgrimage
contract_planetary_pilgrimage_5 = {
	category = planetary_pilgrimage_special
	desc = "contract_planetary_pilgrimage_5_desc"
	lore_issued = "contract_planetary_pilgrimage_5_lore_issued"
	lore_completed = "contract_planetary_pilgrimage_5_lore_completed"
	lore_failed = "contract_planetary_pilgrimage_5_lore_failed"
	event_chain = contract_planetary_pilgrimage_5_chain
	icon = gfx/interface/icons/situation_log/situation_log_quest.dds
	picture = GFX_evt_tiyanki_observation
	location = yes
	time_to_accept = @contract_time_base

	possible_operator = {
		custom_tooltip = {
			fail_text = already_doing_same_contract
			NOT = {
				has_event_chain = contract_planetary_pilgrimage_5_chain
			}
		}
		contract_target_system_reachable = {
			SYSTEM = from
		}
	}

	potential_issuer = {
		always = no
	}

	abort_trigger = {
		NAND = {
			has_event_chain = contract_planetary_pilgrimage_5_chain
			contract_target_system_reachable = {
				SYSTEM = fromfrom
			}
		}
		fromfrom = {
            NOT = { is_pilgrimage_site_space_fauna = yes }
            is_point_of_interest = {
                id = pilgrimage_poi@this
                event_chain = contract_planetary_pilgrimage_5_chain
                owner = root
            }
        }
	}

	counter = {
		contract_progress_counter = {
			max = 1
		}
	}

	on_issue = {
		fromfrom = {
			set_star_flag = planetary_pilgrimage_contract_issued@root
		}
	}

	on_accept = {
		begin_event_chain = {
			event_chain = contract_planetary_pilgrimage_5_chain
			target = ROOT
		}
		fromfrom = {
			limit = {
				is_pilgrimage_site_space_fauna = yes
			}
			root = {
				create_point_of_interest = {
					id = pilgrimage_poi@prev
					name = "pilgrimage_poi_name"
					desc = "pilgrimage_poi_desc"
					event_chain = contract_planetary_pilgrimage_5_chain
					location = prev
				}
			}
			enable_special_project = {
				name = "CONTRACT_PILGRIMAGE_5_PROJECT"
				location = fromfrom
				owner = root
			}
		}
	}

	on_stop = {
		end_event_chain = contract_planetary_pilgrimage_5_chain
		fromfrom = {
			remove_star_flag = planetary_pilgrimage_contract_issued@root
		}
	}

	on_success = {
		if = {
			limit = {
				exists = root
			}
			planetary_pilgrimage_scaling_reward_effect = {
				UNITY = 1000
				RESOURCE = physics_research
				RESOURCE_AMOUNT = 500
			}
			hidden_effect = {
				country_event = { id = EaC_civic_mechanic_events.5006 days = 1 }
				country_event = { id = nomads.6998 days = 1 }
			}
		}
	}

	on_cancel = {
		if = {
			limit = { has_event_chain = contract_planetary_pilgrimage_5_chain }
			end_event_chain = contract_planetary_pilgrimage_5_chain
		}
		if = {
			limit = {
				has_special_project = CONTRACT_PILGRIMAGE_5_PROJECT
			}
			abort_special_project = {
				type = CONTRACT_PILGRIMAGE_5_PROJECT
			}
		}
	}
}
