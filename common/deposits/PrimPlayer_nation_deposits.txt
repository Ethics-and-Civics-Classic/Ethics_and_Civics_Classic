## Possible Deposit Variables ##
# resources, resource and amount
# Potential trigger (planet scope)
# planet_modifier - applied to planet only when deposit blocker has been cleared
# constant_modifier - applied to planet always
# station = station class in orbit to gather
# ai_weight = AI willingness to clear blocker. Note: weight 1 is default and is treated specially - 
# It is set to weight 0 if the planet has no need of more max districts or of districts
# Of the type gained from clearing the blocker.

# Provided on homeworlds at game-start if you have the civic-origin 'Lost Colony'
d_crashed_colony_ship = {
	is_for_colonizable = yes
	icon = "d_crashed_ship"
	category = deposit_cat_rare

	potential = { always = no }

	planet_modifier = {
		job_broken_shackles_scavenger_per_pop = 0.04
		pop_environment_tolerance = 0.30
	}
	triggered_planet_modifier = {
		potential = {
			is_artificial = no
		}
		district_farming_max_add = 1
		district_generator_max_add = 1
	}
}

# The Obelisk is generated on homeworlds at game-start if you have the civic-origin 'The Obelisk'
d_PrimPlayer_the_obelisk_deposit = {
	icon = d_monument

	is_for_colonizable = yes
	category = deposit_obelisk

	potential = { always = no }

	planet_modifier = {
		
	}
	
	# Primitive & Normal Empire
	triggered_planet_modifier = {
		potential = {
			exists = owner
			owner = {
				OR = {
					any_situation = { is_situation_type = PrimPlayer_situation } # Much more efficient trigger
					AND = {
						is_primitive = yes
						OR = {
							has_country_flag = machine_age
							has_country_flag = industrial_age
							has_country_flag = atomic_age
							has_country_flag = early_space_age
						}
					}
				}
				is_gestalt = no
			}
		}
		modifier = {
			job_primitive_researcher_2_add = 200
			job_primitive_priest_2_add = 200
		}
	}
	
	# Primitive & Hive Empire
	triggered_planet_modifier = {
		potential = {
			exists = owner

			}
		}
		modifier = {
			job_primitive_hive_brain_drone_add = 200
		}
	}
	
	# Primitive & Machine Empire
	triggered_planet_modifier = {
		potential = {
			exists = owner

					}
				}
				is_machine_empire = yes
			}
		}
		modifier = {
			job_primitive_machine_calculator_add = 200
		}

		weight = 0

	}

		weight = 0
	}

# Decrepit Dwellings are generated on homeworlds at game-start
d_ethic_fanatic_authoritarian_nation_blocker = {
	icon = d_slums
	time = 180

	is_for_colonizable = yes
	category = deposit_cat_nation_blocker

	potential = { always = no }

	planet_modifier = {
		planet_max_districts_add = -2
	}

	on_cleared = {
		primitive_nation_absorption_effect = yes
		hidden_effect = { 
			planet = {
				remove_carrier_flag = PrimPlayer_authoritarian_union_situation_complete
			}
		}
	}

	inline_script = deposits/ai_weight_pop_creating_blocker

	resources = {
		category = deposit_blockers
		cost = {
			trigger = { planet = { NOT = { is_planet_class = pc_nuked } } }
			unity = 500
		}
		cost = {
			trigger = { planet = { is_planet_class = pc_nuked } }
			minerals = 200
		}
	}
	
	can_be_cleared = {
		custom_tooltip = {
			fail_text = has_absorbed_ethic_fanatic_authoritarian_nation
			planet = {
				OR = {
					has_carrier_flag = PrimPlayer_authoritarian_union_situation_complete
					NOT = { any_country = { has_country_flag = ethic_fanatic_authoritarian_nation@root.owner } }
				}
			}
		}
	}

}
