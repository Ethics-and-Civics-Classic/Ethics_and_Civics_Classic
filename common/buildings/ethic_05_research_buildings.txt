building_nomad_conservatory = {
	icon = holding_park_ranger_lodge
	planet_limit = 1
	base_buildtime = @b1_time

	category = research

	building_sets = {
		urban
		research
		society
		unity
		ark_pilgrims
	}

	potential = {
		NOT = {	has_modifier = slave_colony	}
		owner? = {
			is_regular_empire = yes
			is_nomadic = yes
			is_green = yes
		}
	}

	allow = {
		custom_tooltip = {
			fail_text = no_free_district_for_ranger_lodge
			planet = {
				num_free_districts = {
					type = any
					value > 0
				}
			}
		}
	}

	destroy_trigger = {
		owner? = {
			OR = {
				owner = { 
					OR = {
						is_regular_empire = no 
						is_nomadic = no
						is_green = no
					}
				}
				has_modifier = slave_colony
			}
		}
	}

	ai_resource_production = {
		unity = 1.0
		mult = value:natural_blocker_count
	}

	ai_resource_production = {
		unity = 0.5
		trigger = { owner_has_resolution = { RES = resolution_ecology_integrated_gardens } }
		mult = value:natural_blocker_count
	}

	ai_resource_production = {
		unity = 0.5
		trigger = { owner_has_resolution = { RES = resolution_ecology_environmental_control_board } }
		mult = value:natural_blocker_count
	}

	ai_resource_production = {
		unity = 0.5
		trigger = { owner_has_resolution = { RES = resolution_ecology_paradise_initiative } }
		mult = value:natural_blocker_count
	}

	# We need to cheat here and tell AI that this will probably give some complex output since AI can not evaluate what this building does without the pops which it cant get without the building - chicken and egg
	ai_resource_production = {
		unity = 10
		society_research = 10

		trigger = {
			NOT = { has_building = building_nomad_conservatory }
		}
	}

	ai_weight_coefficient = 1.2

	# Here we try to adress the second issue that the AI cant evaluate the pop job output of the bio trophy given the happiness trigger that was put inplace to guard against players exploiting bio trophies

	#science
	ai_resource_production = {
		unity = 10
		society_research = 10

		trigger = {
			planet_resource_compare = {
				resource = society_research
				value >= 100
				type = produces
			}
		}
	}

	#energy
	ai_resource_production = {
		unity = 10
		energy = 10

		trigger = {
			planet_resource_compare = {
				resource = energy
				value >= 100
				type = produces
			}
		}
	}
	
	planet_modifier = {
		job_biologist_add = 100 # 100 Gardeners no matter what
		deposit_blockers_natural_unity_produces_add = 1.0
	}

	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		job_biologist_add = 100 # +100 Gardeners per natural blocker
		mult = value:natural_blocker_count
	}

	triggered_planet_modifier = {
		potential = {
			owner_has_resolution = { RES = resolution_ecology_paradise_initiative }
		}
		job_biologist_add = 100 # +100 Gardeners per natural blocker if GalCom approves
		mult = value:natural_blocker_count
	}


	triggered_planet_modifier = {
		potential = { owner_has_resolution = { RES = resolution_ecology_integrated_gardens } }
		deposit_blockers_natural_unity_produces_add = 0.5
	}

	triggered_planet_modifier = {
		potential = { owner_has_resolution = { RES = resolution_ecology_environmental_control_board } }
		deposit_blockers_natural_unity_produces_add = 1.0
	}

	triggered_planet_modifier = {
		potential = { owner_has_resolution = { RES = resolution_ecology_paradise_initiative } }
		deposit_blockers_natural_unity_produces_add = 1.5
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = exotic_gases
			COST = @b1_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		upkeep = {
			energy = @b1_upkeep
		}
	}

	triggered_desc = {
		text = biologist_is_ark_gardener_tt
	}

	on_built = {
		planet = {
			add_blocker = { type = d_ark_nature_preserve_blocker }
		}
	}

	on_destroy = {
		planet = {
			if = {
				limit = { has_deposit = d_ark_nature_preserve_blocker }
				every_deposit = {
					limit = {
						is_deposit_type = d_ark_nature_preserve_blocker
					}
					remove_deposit = yes
				}
			}
		}
	}
}
