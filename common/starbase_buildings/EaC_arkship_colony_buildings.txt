	construction_days = 360
	category = sm_colony_cat

	}

			}

		}
	}

		OR = {
			owner? = {

			}

					has_modifier = slave_colony
					has_modifier = resort_colony
				}
			}
		}
	}

	destroy_trigger = {
		OR = {
			owner? = {
				is_regular_empire = no
			}
			colony? = {
				OR = {
					has_modifier = slave_colony
					has_modifier = resort_colony
				}
			}
		}
	}

	show_in_tech = "tech_galactic_markets"

	show_tech_unlock_if = {
		is_nomadic = yes
		is_regular_empire = yes

		category = starbase_arkship_upgrades

			alloys = @arkship_cost_alloys_200
			rare_crystals = @arkship_cost_strategic_300
		}

		}
	}

	planet_modifier = {

	ai_weight = {

			}
