	desc = {
		trigger = {
			has_policy_flag = first_contact_proactive
		}

	}
	desc = {
		trigger = {
			NOT = { has_policy_flag = first_contact_proactive }
		} 
		text = "ethic_civic_classic_flag.4.desc"
	}
	
	option = {
		name = "ethic_civic_classic_flag.4.full.proactive"
		trigger = { has_policy_flag = first_contact_proactive }
		destroy_situation = FROM
		add_resource = { unity = 1000 }
	}
	
	option = {
		name = "ethic_civic_classic_flag.4.full"
		trigger = { NOT = { has_policy_flag = first_contact_proactive } }
		if = {
			limit = { capital_scope = { NOT = { has_modifier = culture_shock_diplomacy } } }
			capital_scope = { 
				add_modifier = {
					modifier = culture_shock_diplomacy
					years = 20
					clear_on_owner_change = yes
				}
			}
		}
		destroy_situation = FROM
	}
	
	option = {
		name = "ethic_civic_classic_flag.4.suppress"
		trigger = {
			OR = {
				has_ethic = ethic_fanatic_authoritarian
				has_ethic = ethic_authoritarian
				is_authoritarian = no
			}
		}
		allow = {
			if = {
				limit = { has_ethic = ethic_fanatic_authoritarian }
				has_country_resource = { type = influence amount >= 50 }
			}
			if = {
				limit = { has_ethic = ethic_authoritarian }
				has_country_resource = { type = influence amount >= 75 }
			}
			else = {
				has_country_resource = { type = influence amount >= 100 }
			}
			custom_tooltip = {
				text = civic_tooltip_not_egalitarian_or_xenophile
				NOR = {
					is_egalitarian = yes 
					is_xenophile = yes
				}
			}
			capital_scope = {
				if = { 
					limit = { has_branch_office = yes }
					hidden_trigger = { branch_office_owner = { has_country_flag = ethic_fanatic_capitalism_nation@root } }
				}
				else = { has_branch_office = no }
			}
		}
		if = {
			limit = { has_ethic = ethic_fanatic_authoritarian }
			add_resource = { influence = -50 }
		}
		else_if = {
			limit = { has_ethic = ethic_authoritarian }
			add_resource = { influence = -75 }
		}
		else = {
			add_resource = { influence = -100 }
		}
		if = {
			limit = { capital_scope = { has_modifier = culture_shock_diplomacy } }
			capital_scope = { remove_modifier = culture_shock_diplomacy }
		}
		hidden_effect = { 
			destroy_situation = FROM
			start_situation = { type = PrimPlayer_awareness_situation target = ROOT }
		}
		custom_tooltip = PrimPlayer_awareness_situation_suppress
	}
	
	after = {
		hidden_effect = {
			if = {
				limit = { any_situation = { is_situation_type = PrimPlayer_awareness_situation } }
				random_situation = { 
					limit = { is_situation_type = PrimPlayer_awareness_situation } 
					hidden_effect = { 
						add_situation_progress = 89 
					}
					set_situation_approach = PrimPlayer_awareness_situation_suppress
				}
			}
			FROM = { remove_situation_flag = PrimPlayer_awareness_situation_event }
		}
	}
}
